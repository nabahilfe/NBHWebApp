
    // Observer für alle hidden Inputs - auswählen z. B. mit data-attribut für Klarheit
    const hiddenFields = document.querySelectorAll('input[type="hidden"][data-observe]');
    const observer = new MutationObserver((mutations) => {
      mutations.forEach((mutation) => {
        if (mutation.type === 'attributes' && mutation.attributeName === 'value') {
          const field = mutation.target;
          const id = field.id || field.name;
          console.log(`Field ${id} changed to:`, field.value);
          // dispatchEvent('change') für Legacy-Listener
          field.dispatchEvent(new Event('change'));
        }
      });
    });

    // Observer für jedes Field aktivieren
    hiddenFields.forEach(field => {
      observer.observe(field, { attributes: true });
    });
